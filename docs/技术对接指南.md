## 1. 腾讯云验证码配置

### 1.1 获取腾讯云验证码

1. 访问 [腾讯云验证码控制台](https://console.cloud.tencent.com/captcha)
2. 点击「新建验证」
3. 填写验证信息：
   - 应用名称：小说平台
   - 应用场景：Web端
   - 验证码类型：滑块验证
4. 点击「确定」创建

### 1.2 配置域名白名单

在验证码详情页，找到「域名配置」：
```
添加白名单：
  - localhost (本地测试)
  - your-domain.com (你的生产域名)
  - your-app.up.railway.app (Railway域名)
```

在验证码详情页可以看到：
```

```

## 2. 邮件发送配置

### 2.1 获取 Resend API Key

1. 访问 [Resend官网](https://resend.com)
2. 注册/登录账号
3. 点击「API Keys」→「Create API Key」
4. 复制你的 API Key

### 2.2 在 Railway 中配置

1. 访问 [Railway控制台](https://railway.app)
2. 进入你的项目 → Variables
3. 添加环境变量：
   ```
   RESEND_API_KEY=你的Resend_API_Key
   ```

### 2.3 验证配置

启动服务器后，测试发送验证码：
```
访问 /payment，注册时输入邮箱，点击获取验证码
```

### 3.1 方案一：出版社对接

#### 步骤1：联系出版社

推荐联系的出版社：
- 阅文集团（起点中文网）

#### 步骤2：提交资质

需要准备：
- 营业执照
- ICP备案
- 公司公章
- 运营计划

#### 步骤3：签订合同

签订版权授权协议，约定：
- 分成比例
- 结算周期
- 版权范围

### 3.2 方案二：第三方API平台

推荐平台：

| 平台 | 价格 | 特点 |
|------|------|------|
| 聚合数据 | 按次计费 | 数据稳定 |
| 天行数据 | 免费+付费 | 文档完善 |
| 阿凡达数据 | 包月制 | 更新及时 |

#### 示例：集成聚合数据API

```javascript
// routes/books.js
const JUHE_API_KEY = process.env.JUHE_API_KEY || '你的API密钥';

async function fetchBooksFromAPI(keyword) {
  const response = await fetch(
    `https://api.juheapi.com/novel/search?keyword=${keyword}&key=${JUHE_API_KEY}`
  );
  
  const data = await response.json();
  
  if (data.error_code === 0) {
    return data.result;
  } else {
    throw new Error(data.reason);
  }
}

app.get('/api/books/search', async (req, res) => {
  try {
    const keyword = req.query.keyword;
    const books = await fetchBooksFromAPI(keyword);
    res.json({ success: true, books });
  } catch (error) {
    res.status(500).json({ success: false, message: error.message });
  }
});
```

          method: 'POST',
          body: formData

        alert('上传失败：' + error.message);
      }
    });
  </script>

</html>

#### 步骤2：后端处理

创建 `routes/upload.js`：
```javascript
const express = require('express');
const router = express.Router();
const multer = require('multer');
const { booksDB, chaptersDB } = require('../db/init');
const { v4: uuidv4 } = require('uuid');
const fs = require('fs');
const path = require('path');

const storage = multer.diskStorage({
  destination: function (req, file, cb) {
    const uploadDir = './uploads';
    if (!fs.existsSync(uploadDir)) {
      fs.mkdirSync(uploadDir, { recursive: true });
    }
    cb(null, uploadDir);
  },
  filename: function (req, file, cb) {
    const uniqueName = Date.now() + '-' + file.originalname;
    cb(null, uniqueName);
  }
});

const upload = multer({ storage: storage });

router.post('/upload', upload.single('file'), async (req, res) => {
  try {
    const { title, author } = req.body;
    const filePath = req.file.path;
    
    // 读取TXT文件内容
    const content = fs.readFileSync(filePath, 'utf8');
    
    // 解析章节（按常见格式）
    const chapters = parseNovelChapters(content);
    
    // 创建书籍
    const bookId = uuidv4();
    const book = {
      id: bookId,
      title: title,
      author: author,
      description: '用户上传的小说',
      cover: null,
      genre: '未知',
      rating: 8.0,
      views: 0,
      status: '完结',
      chapters: chapters.length,
      createdAt: Date.now(),
      updatedAt: Date.now()
    };
    
    booksDB.set(bookId, book);
    
    // 保存章节
    chapters.forEach((chapter, index) => {
      const chapterId = uuidv4();
      chaptersDB.set(chapterId, {
        id: chapterId,
        bookId: bookId,
        chapterNumber: index + 1,
        title: chapter.title,
        content: chapter.content,
        isFree: index < 10, // 前10章免费
        createdAt: Date.now()
      });
    });
    
    // 删除临时文件
    fs.unlinkSync(filePath);
    
    res.json({
      success: true,
      bookId: bookId,
      chapters: chapters.length
    });
  } catch (error) {
    res.status(500).json({
      success: false,
      message: error.message
    });
  }
});

function parseNovelChapters(content) {
  const chapters = [];
  const lines = content.split('\n');
  let currentTitle = '';
  let currentContent = '';
  
  lines.forEach(line => {
    line = line.trim();
    if (!line) return;
    
    // 匹配章节标题（常见格式：第X章 标题 或 第X节 标题）
    const chapterMatch = line.match(/^第[0-9一二三四五六七八九十百千万]+[章节卷].*$/);
    
    if (chapterMatch) {
      // 保存上一章
      if (currentTitle && currentContent) {
        chapters.push({
          title: currentTitle,
          content: currentContent.trim()
        });
      }
      // 开始新章节
      currentTitle = line;
      currentContent = '';
    } else if (currentTitle) {
      currentContent += line + '\n';
    }
  });
  
  // 保存最后一章
  if (currentTitle && currentContent) {
    chapters.push({
      title: currentTitle,
      content: currentContent.trim()
    });
  }
  
  return chapters;
}

module.exports = router;
```

## 5. 常见问题

### Q1: 验证码验证失败怎么办？

A: 检查以下几点：
1. 腾讯云控制台中的域名白名单是否配置正确
2. App Secret Key 是否正确
3. 浏览器控制台是否有错误信息

### Q2: 邮件发送失败怎么办？

A: 检查以下几点：
1. RESEND_API_KEY 是否在 Railway 中正确配置
2. API Key 是否有效
3. Resend 官网是否有错误日志

### Q3: 如何快速获取大量小说数据？

A: 推荐方案：
1. 使用用户上传功能，让用户自己上传
2. 接入第三方API平台
3. 联系出版社获取授权
